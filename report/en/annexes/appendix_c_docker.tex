\chapter{Docker Compose Extracts}

\begin{lstlisting}[language=YAML, caption={PostgreSQL + TimescaleDB service}]
postgres:
  build:
    context: ./infra/postgres
    dockerfile: Dockerfile    # FROM timescale/timescaledb-ha:pg16-all
  container_name: caimp-postgres
  environment:
    POSTGRES_DB:       ${POSTGRES_DB:-caimp}
    POSTGRES_USER:     ${POSTGRES_USER:-caimp}
    POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
  volumes:
    - postgres_data:/var/lib/postgresql/data
  networks: [caimp_net]
  healthcheck:
    test: ["CMD-SHELL", "pg_isready -U caimp -d caimp"]
    interval: 10s
    retries: 10
\end{lstlisting}

\begin{lstlisting}[language=YAML, caption={Correlation Engine service}]
correlation-engine:
  build:
    context: ./services/correlation-engine
  environment:
    DATABASE_URL:     postgresql://caimp:${POSTGRES_PASSWORD}@postgres/caimp
    NATS_URL:         nats://nats:4222
    OLLAMA_URL:       http://ollama:11434
    OLLAMA_LLM_MODEL: ${OLLAMA_LLM_MODEL:-llama3.1:8b-instruct-q4_K_M}
    LOOKBACK_MINUTES: "30"
  ports: ["9095:9095"]
  depends_on:
    postgres: {condition: service_healthy}
    nats:     {condition: service_healthy}
    ollama:   {condition: service_healthy}
    migrate:  {condition: service_completed_successfully}
  networks: [caimp_net]
\end{lstlisting}
